/** Minimal DOM helpers so panels stay readable without a framework. */
type Attrs = Record<string, string | number | boolean | ((ev: Event) => void) | undefined>;
type Child = Node | string | null | undefined | false;

export function h<K extends keyof HTMLElementTagNameMap>(tag: K, attrs: Attrs = {}, ...children: Child[]): HTMLElementTagNameMap[K] {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (v === undefined || v === false) continue;
    if (k.startsWith('on') && typeof v === 'function') el.addEventListener(k.slice(2).toLowerCase(), v as EventListener);
    else if (k === 'class') el.className = String(v);
    else if (k === 'value') (el as HTMLInputElement).value = String(v);
    else if (k === 'checked') (el as HTMLInputElement).checked = v === true;
    else el.setAttribute(k, String(v));
  }
  for (const c of children) {
    if (c === null || c === undefined || c === false) continue;
    el.append(typeof c === 'string' ? document.createTextNode(c) : c);
  }
  return el;
}

**What the site does**

- **3D scene** (Three.js + Vite + TypeScript, no framework): Sun, eight planets with orbit lines, a distance grid in au, and all 42,075 NEOs. Asteroid positions are solved from their own elements and epoch in a vertex shader every frame, so time runs at full frame rate with zero CPU cost for the cloud. Planets, comets, selected bodies, picking and distances use a matching CPU solver.
- **Time**: play/pause, reversible speeds from 1 min/s to 5 yr/s, step buttons, date/time jump, a 1900–2200 scrub bar, keyboard shortcuts.
- **Search and filter**: name/designation search, hazard status, Sentry monitoring, orbit class, absolute magnitude with approximate size, Earth MOID, approach within ±N days of the current time, measured diameters. Colour by hazard, class or size. Live count of visible objects.
- **Object detail**: orbital elements, physical properties, Sentry impact probability and Palermo/Torino values, full close-approach list with click-to-jump, live Sun and Earth distances.
- **Highlighting**: hazardous and Sentry objects stand out, a "close approaches near this date" list, and Earth-to-asteroid lines drawn during approaches.
- **Comets**: all 4,068, with elliptic, parabolic and hyperbolic propagation from perihelion time, orbit lines clipped at 60 au, and anti-sunward tails.
- **Focus and follow** camera, and **deep links** that encode time, selection, camera, speed, playback and follow state in the URL hash.
